## AIM   
To write a C program to implement a symbol table.

## ALGORITHM
1.	Start the program.
2.	Get the input from the user with the terminating symbol ‘$’.
3.	Allocate memory for the variable by dynamic memory allocation function.
4.	If the next character of the symbol is an operator then only the memory is allocated.
5.	While reading, the input symbol is inserted into symbol table along with its memory address.
6.	The steps are repeated till ‘$’ is reached.
7.	To reach a variable, enter the variable to be searched and symbol table has been checked for corresponding variable, the variable along with its address is displayed as result.
8.	Stop the program. 

## PROGRAM
```
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct symbol
{
    char name[20];
    void *address;
};

int main()
{
    struct symbol table[50];
    int count = 0, i;
    char input[20], search[20];

    printf("Enter symbols (Enter $ to stop):\n");

    while (1)
    {
        scanf("%s", input);

        if (strcmp(input, "$") == 0)
            break;

        strcpy(table[count].name, input);
        table[count].address = malloc(sizeof(char));

        count++;
    }

    printf("\nSymbol Table\n");
    printf("-----------------------------\n");
    printf("Symbol\t\tAddress\n");
    printf("-----------------------------\n");

    for (i = 0; i < count; i++)
    {
        printf("%s\t\t%p\n", table[i].name, table[i].address);
    }

    printf("\nEnter symbol to search: ");
    scanf("%s", search);

    for (i = 0; i < count; i++)
    {
        if (strcmp(search, table[i].name) == 0)
        {
            printf("Symbol Found\n");
            printf("Name: %s\nAddress: %p\n", table[i].name, table[i].address);
            break;
        }
    }

    if (i == count)
        printf("Symbol not found.\n");

    for (i = 0; i < count; i++)
        free(table[i].address);

    return 0;
}
```
